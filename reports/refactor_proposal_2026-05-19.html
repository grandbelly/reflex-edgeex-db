5-4. TimescaleDB 청크 튜닝

-- 현재 chunk_time_interval 확인
SELECT hypertable_name, chunk_time_interval FROM timescaledb_information.dimensions;

-- RPI SD card I/O 최소화: 1일 청크 → 7일 청크로
SELECT set_chunk_time_interval('influx_hist', INTERVAL '7 days');

-- 압축 정책 (older than 7d compressed)
ALTER TABLE influx_hist SET (timescaledb.compress);
SELECT add_compression_policy('influx_hist', INTERVAL '7 days');

  • statsforecast, numpy, pandas 가 가장 큼 — wheel 캐시 활용
